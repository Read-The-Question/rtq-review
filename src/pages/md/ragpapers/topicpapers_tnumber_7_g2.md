&= \dfrac {20} {4 \times 1000} \\\\
&= \dfrac {5 \times 4} {1000 \times 4} \\\\
&= \dfrac {5 \times \cancel {4}} {1000 \times \cancel {4}} \\\\
&= \dfrac {5} {1000} \\\\
&= 0.005
\end{aligned}
