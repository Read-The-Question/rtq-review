                                &= \dfrac {2 \times 5 \times 11} {2}\\\\
                                &= \dfrac {\cancel {2} \times 5 \times 11} {\cancel {2}} \\\\
                                &= 5 \times 11 \\\\
                                &= 55
\end {aligned}
