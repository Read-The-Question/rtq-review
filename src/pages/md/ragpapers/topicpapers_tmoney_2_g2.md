$
\begin{aligned}
a + 1.3a                        &= 46 \\\\
2.3a                            &= 46 \\\\

a                               &= \dfrac {\cancel{23} \times 2 \times 10} {\cancel{23}} \\\\
a                               &= 2 \times 10 \\\\
a                               &= \pounds 20 \\\\
\text {Jon has}                 &= 1.3 \times a \\\\
                                &= 1.3 \times 20 \\\\
                                &= \pounds 26
\end{aligned}
$
