$
\begin{aligned}
\text{Final Bill}       &= \text{Total bill} + \text{VAT} \\\\

                        &= 276.60 + (5 \% \times 276.60) \\\\
                        &= 276.60 + (\dfrac {5} {100} \times 276.60) \\\\
                        &= 276.60 + (\dfrac {276.60 \times 5}{100}) \\\\
                        &= 276.60 + \dfrac {1383} {100} \\\\
                        &= 276.60 + 13.83 \\\\
                        &= \pounds 290.43
\end{aligned}
$
