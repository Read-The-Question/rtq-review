\begin{aligned}
600 \div a              &= 50 \\\\
\dfrac {600} {a}        &= 50 \\\\

a                       &= \dfrac {600} {50}  \\\\
a                       &= \dfrac {600} {25 \times 2}  \\\\
a                       &= \dfrac {24} {2}  \\\\
a                       &= \dfrac {12 \times 2} {2}  \\\\
a                       &= \dfrac {12 \times \cancel{2}} {\cancel{2}}  \\\\
a                       &= 12
\end{aligned}
$

</div>
