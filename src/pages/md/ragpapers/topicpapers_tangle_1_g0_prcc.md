![missing image](/papers/missing_image.svg)

%format% - a = 58o, b = 53o
%format% - a = 58o, b = 63o

$
\begin{aligned}
b + 69                                                  &= 132 \\\\
b                                                       &= 132 - 69 \\\\
b                                                       &= 63 \degree
\end{aligned}
$

$
\begin{aligned}
a + b + 69                                              &= 180 \\\\
a + 63 + 69                                             &= 180 \\\\
a + 132                                                 &= 180 \\\\
a                                                       &= 180 - 132 \\\\
a                                                       &= 48 \degree
\end{aligned}
$
