s ^ 2 + 3t      &= 10
\end{aligned}

Let's verify for $1, 3$.

$
\begin{aligned}

&= 1 ^ 2 + 3 \times 3 \\\\
&= 1  + 9 \\\\
&= 10
\end{aligned}

\begin{aligned}
&= s ^ 2 + 3t \\\\
&= 2 ^ 2 + 3 \times 2 \\\\

$\dfrac{3}{4} + 0.35$ 

- $\dfrac{6}{5}$
- $\dfrac{11}{10}$
- $\dfrac{65}{4}$
- $\dfrac{1}{10}$
- $\dfrac{4}{5}$

</div>
<div class='workings'>
<div class='working'>

$
\begin{aligned}
0.35 \text{is the decimal equivalent of} \\\\
&= \dfrac{35}{100} \\\\
&= \dfrac{5 \times 7}{5 \times 20} \\\\
&= \dfrac{\cancel 5 \times 7}{\cancel 5 \times 20} \\\\
&= \dfrac{7}{20} \\\\
\text{then,} \\\\
&= \dfrac{3}{4} + 0.35 \\\\
&= \dfrac{3}{4} + \dfrac{7}{20} \\\\
&= \dfrac{3\times5}{4\times5} + \dfrac{7}{20} \\\\
&= \dfrac{15}{20} + \dfrac{7}{20} \\\\
&= \dfrac{15+7}{20} \\\\
&= \dfrac{22}{20} \\\\
&= \dfrac{2 \times 11}{2\times 10} \\\\
&= \dfrac{\cancel 2 \times 11}{ \cancel 2\times 10} \\\\
&= \dfrac{11}{10}
\end{aligned}
$

</div>

</div>
