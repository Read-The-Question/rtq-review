                                  &= 72 \degree \\\\
\text{Number of children}         &= \dfrac {72} {360} \times 40 \\\\
                                  &= \dfrac {72 \times 40} {360} \\\\
                                  &= \dfrac {8 \times 9 \times 40} {9 \times 40} \\\\
                                  &= \dfrac {8 \times \cancel 9 \times \cancel {40}} {\cancel 9 \times \cancel {40}} \\\\
                                  &= 8
\end{aligned}
$
