$
\begin{aligned}

                                    &= \dfrac {6 \times 126} {6} \\\\
                                    &= \dfrac {\cancel 6 \times 126} {\cancel 6} \\\\
                                    &= 126 \ \text{cm}
\end{aligned}
$
