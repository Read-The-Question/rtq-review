<div class='question subquestion'>

$\boxed { \quad\dfrac{31}{40} = \text{\textunderscore \textunderscore \textunderscore} \quad}$

</div>
<div class='workings'>
<div class='working'>

$
\begin {aligned}
&= \dfrac {31} {40} \\\\
&= \dfrac {29 + 2} {40} \\\\
&= \dfrac {29} {40} + \dfrac {2} {40} \\\\
&= 0.725 + \dfrac {2} {20 \times 2} \\\\
&= 0.725 + \dfrac {\cancel {2} } {20 \times \cancel {2} } \\\\
&= 0.725 + \dfrac {1} {20} \\\\
&= 0.725 + 0.05 \\\\
&= 0.775
\end {aligned}
$

</div>
</div>
<div class='answers'>
<div class='answer'>

$0.775$

</div>

&= 0.675 - \dfrac {1} {10} \\\\
&= 0.675 - 0.1 \\\\
&= 0.575
\end {aligned}
