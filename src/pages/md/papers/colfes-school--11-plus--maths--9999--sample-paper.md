                                &= \dfrac {5} {10} \\\\
                                &= \dfrac {5} {5 \times 2} \\\\
                                &= \dfrac {\cancel {5}} {\cancel {5} \times 2} \\\\
                                &= \dfrac {1} {2}
\end {aligned}
$

                        &= \dfrac {\cancel {2} \times 11} {\cancel {2}} + \dfrac {\cancel {2} \times 11} {\cancel {2}} - 5 + 22 \\\\
                        &= 11 + 11 - 5 + 22 \\\\
                        &= 44 - 5 \\\\
                        &= 39
\end {aligned}
