&= \dfrac {28} {35} - \dfrac {21} {35} \\\\
&= \dfrac {28 - 21} {35} \\\\
&= \dfrac {7} {35} \\\\
&= \dfrac {\cancel {7}} {\cancel {7} \times 5} \\\\
&= \dfrac {1} {5}
\end{aligned}
