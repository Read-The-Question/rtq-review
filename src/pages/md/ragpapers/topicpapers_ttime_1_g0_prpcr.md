<div class='question question'>

</div>

<div class='working'>

\begin{aligned}
\text{Time to leave}   &= 8{:}15 \ \text{am} - 40 \ \text{minutes} \\\\
                       &= 7{:}35 \ \text{am}
\end{aligned}
$

</div>
</div>

\end{aligned}
