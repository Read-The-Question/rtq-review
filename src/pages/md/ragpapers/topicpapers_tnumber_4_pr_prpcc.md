<li>
<div class='question_envelope rag_not_found rag_not_found subquestion'>
<div class='question subquestion'>

$8200 \times 107$

</div>
<div class='workings'>
<div class='working'>

$
\begin{aligned}
&= 8200 \times 107 \\\\
&= 82 \times 107\times 100 \\\\
&= 8774 \times 100 \\\\
&= 877400
\end{aligned}
$

</div>
</div>
<div class='answers'>
<div class='answer'>

$877400$

</div>
</div>

</div>
</li>
<li>
<div class='question_envelope rag_not_found rag_not_found subquestion'>
<div class='question subquestion'>

$8774 \div 107$

</div>
<div class='workings'>
<div class='working'>

$
\begin{aligned}
&= 8774 \div 107 \\\\
&= \dfrac{8774}{107} \\\\
&= \dfrac {82 \times 107} {107} \\\\
&= \dfrac {82 \times \cancel {107}} {\cancel {107}} \\\\
&= 82
\end{aligned}
$

</div>
</div>
<div class='answers'>
<div class='answer'>

$82$

</div>
</div>

</div>
</li>
<li>
<div class='question_envelope rag_not_found rag_not_found subquestion'>
<div class='question subquestion'>

$8774 \div 41$

</div>
<div class='workings'>
<div class='working'>

$
\begin{aligned}
&= \dfrac{8774}{41} \\\\
&= \dfrac {8774 \times 2}{41 \times 2} \\\\
&= \dfrac {8774 \times 2}{82} \\\\
&= \dfrac {82 \times 107 \times 2} {82} \\\\
&= \dfrac {\cancel{82} \times 107 \times} {\cancel {82}} \\\\
&= 107 \times 2 \\\\
&= 214
\end{aligned}
$

</div>
</div>
<div class='answers'>
<div class='answer'>

$214$

</div>
</div>
