</div>
<div class='workings'>
<div class='working'>

$
\begin{matrix}
    \footnotesize{\maroonC
{( \dfrac {5} {8} \times \dfrac{1} {1} )}} 
&&  \footnotesize{\maroonC
{( \dfrac {5} {8} \times \dfrac{2} {2} )}} 
&&  \footnotesize{\maroonC
{( \dfrac {5} {8} \times \dfrac{3} {3} )}}  
&&  \footnotesize{\maroonC
{( \dfrac {5} {8} \times \dfrac{4} {4} )}} \\\\

\begin{aligned}
\text{Next fraction}        &= \dfrac {5} {8} \times \dfrac{4} {4} \\\\
                            &= \dfrac {5 \times 4} {8 \times 4} \\\\
                            &= \dfrac {20} {32}
\end{aligned}
