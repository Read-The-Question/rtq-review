\begin{matrix}
&   \footnotesize{\maroonC
{(\times \dfrac{9} {10})}} 
&&  \footnotesize{\maroonC

\end{matrix}

\begin{aligned}
\text{Missing number}   &= 660 \times \dfrac{9} {10} \\\\
                        &= \dfrac{660 \times 9} {10} \\\\
                        &= \dfrac{66 \cancel {0} \times 9} {1 \cancel {0}} \\\\
                        &= 66 \times 9 \\\\
                        &= 594
\end{aligned}
