$
\begin{aligned}
a \times 7  &=  896 \\\\ 
a           &=  \dfrac {896} {7} \\\\
a           &=  \dfrac {128 \times 7} {7} \\\\
a           &=  \dfrac {128 \times \cancel{7}} {\cancel{7}} \\\\
a           &=  128
\end{aligned}
