      &= \dfrac {4} {2 \times 4} - \dfrac {1} {8} \\\
      &= \dfrac {4} {8} - \dfrac {1} {8} \\\\
      &= \dfrac {4 - 1} {8} \\\\
      &= \dfrac {3} {8} \\\\
A + B &= \dfrac {1} {4} + \dfrac {3} {8} \\\\

      &= \dfrac {2} {8} + \dfrac {3} {8} \\\\
      &= \dfrac {2 + 3} {8} \\\\
      &= \dfrac {5} {8} 
\end{aligned}

      &= \dfrac {1} {8} \times \dfrac {1} {2} \\\\
      &= \dfrac {1} {8 \times 2} \\\\
      &= \dfrac {1} {16}
\end{aligned}

$B + F + E + D$?

</div>
<div class='workings'>
<div class='working'>

$
\begin{aligned}
F+E+D       &= \dfrac {1}{4} \\\\
B           &= \dfrac {3}{8} \\\\
B+F+E+D     &= \dfrac {1}{4} + \dfrac {3}{8} \\\\
            &= \dfrac {1 \times 2}{4 \times 2} + \dfrac {3}{8} \\\\
            &= \dfrac {2}{8} + \dfrac {3}{8} \\\\
            &= \dfrac {2+3}{8} \\\\
            &= \dfrac {5} {8}
\end{aligned}
$
