                                   &= 1 - \dfrac {4}{9} \\\\
                                   &= \dfrac {1 \times 9} {1 \times 9} - \dfrac {4}{9} \\\\
                                   &= \dfrac {9-4} {9} \\\\
                                   &= \dfrac {5} {9}
\end{aligned}
