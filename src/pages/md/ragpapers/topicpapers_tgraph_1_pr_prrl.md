\end{aligned}

                                &= \sqrt {0 + 16} \\\\
                                &= \sqrt {16} \\\\
                                &= 4 \ \text{cm} \\\\
\text{Breadth of rectangle}     &= \sqrt {(-1 - 4) ^ 2 + (1 - 1) ^ 2} \\\\

                                &= \sqrt {25} \\\\
                                &= 5 \ \text{cm} \\\\
\text{Area of rectangle}        &= \text{Length of rectangle} \times \text{Breadth of rectangle} \\\\
                                &= 4 \times 5 \\\\
