<div class='workings'>
<div class='working'>

$
\begin{aligned}
&= 207.2 \div 56 \\\\
&= \dfrac {207.2} {56} \\\\
&= \dfrac {207.2 \times 10} {56 \times 10} \\\\
&= \dfrac {2072} {56 \times 10} \\\\
&= \dfrac {37 \times 56} {56 \times 10} \\\\
&= \dfrac {37 \times \cancel{56}} {\cancel{56} \times 10} \\\\
&= \dfrac {37} {10} \\\\
&= 3.7
\end{aligned}
$
