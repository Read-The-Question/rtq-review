$\dfrac{2}{3} - \dfrac{1}{6}$

</div>

<div class='working'>

$
\begin{aligned}
&= \dfrac {2} {3} - \dfrac {1} {6} \\\\
&= \dfrac {2 \times 2} {3 \times 2} - \dfrac {1} {6} \\\\
&= \dfrac {4} {6} - \dfrac {1} {6} \\\\
&= \dfrac {4 - 1} {6} \\\\
&= \dfrac {3} {6} \\\\
&= \dfrac {1} {2}
\end{aligned}
$

</div>
</div>
<div class='answers'>
<div class='answer'>

$\dfrac {1} {2}$

&= 2 \times 45  \\\\
&= 90
\end{aligned}
$

</div>
</div>
<div class='answers'>
<div class='answer'>

$90$

\end{aligned}
