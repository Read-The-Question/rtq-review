$
\begin{aligned}

\end{aligned}
$

From Row 2, where $x = 4$,

$
\begin{aligned}
a \times 4 + b &= 7 \\\\
4a + b &= 7 \\\\
\end{aligned}

$
\begin{aligned}
4a + b &= 7 \\\\

\end{aligned}

\begin{aligned}

\end{aligned}
