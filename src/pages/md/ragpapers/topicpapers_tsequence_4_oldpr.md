$
\begin {matrix}
&   \footnotesize{\maroonC
{(+ \dfrac {3} {5})}}
&&  \footnotesize{\maroonC
{(+ \dfrac {3} {5})}}
&&  \footnotesize{\maroonC

&&  \footnotesize{\maroonC
{(+ \dfrac {3} {5})}}
&&  \footnotesize{\maroonC
{(+ \dfrac {3} {5})}}  \\

\end {matrix}
$

$
\begin {aligned}
\text{First missing number}   &= 1 \dfrac {4} {5} + \dfrac {3} {5} \\\\
                              &= \dfrac {9} {5} + \dfrac {3} {5} \\\\
                              &= \dfrac {9 + 3} {5} \\\\
                              &= \dfrac {12} {5} \\\\
                              &= 2 \dfrac {2} {5} \\\\
\text{Second missing number}  &= 2 \dfrac {2} {5} + \dfrac {3} {5} \\\\
                              &= \dfrac {12} {5} + \dfrac {3} {5} \\\\
                              &= \dfrac {12 + 3} {5} \\\\
                              &= \dfrac {15} {5} \\\\
                              &= \dfrac {3 \times 5} {5} \\\\
                              &= \dfrac {3 \times \cancel {5}} {\cancel {5}} \\\\
                              &= 3
