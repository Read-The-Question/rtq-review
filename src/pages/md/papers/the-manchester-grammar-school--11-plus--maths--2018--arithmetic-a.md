&= 3 \dfrac {3} {8} - 1 \dfrac {3} {4} \\\\

&= \dfrac {27} {8} - \dfrac {14} {8} \\\\
&= \dfrac {27 - 14} {8} \\\\
&= \dfrac {13} {8} \\\\
&= 1\dfrac {5} {8}
\end{aligned}

\end{aligned}

$
\begin{aligned}

\end{aligned}

2c + 4b - 2c - 3b       &= 100 \\\\

\end{aligned}
