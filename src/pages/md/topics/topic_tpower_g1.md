$
\begin{matrix}
&   \footnotesize {\maroonC {(-3)} } 
&&  \footnotesize {\maroonC {(-3)} }  
&&  \footnotesize {\maroonC {(-3)} }   
&&  \footnotesize {\maroonC {(-3)} }   
&&  \footnotesize {\maroonC {(-3)} }
&&  \footnotesize {\maroonC {(-3)} }
&&  \footnotesize {\maroonC {(-3)} }
&&  \footnotesize {\maroonC {(-3)} }  \\
28,  &&  25,  &&  22,  &&  19,  &&  16,  &&  13,  &&  10,  && \ldots,  &&  \ldots
\end{matrix}
$

$
\begin{aligned}
\text{First missing number}       &= 10 - 3 \\\\
                                  &= 7 \\\\
\text{Second missing number}      &= 7 - 3 \\\\
                                  &= 4
\end{aligned}
$
