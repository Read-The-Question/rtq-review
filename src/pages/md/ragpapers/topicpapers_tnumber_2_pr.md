\boxed{\phantom{\ 1 \ }} \

(3 - \dfrac{8}{3}) = 24

&= 24
\end{aligned}

a                                               &= \dfrac {25 \times 11 \times 27 \times 20} {25 \times 20} \\\\
a                                               &= \dfrac {27 \times \cancel{20} \times 11 \times \cancel{25}} {\cancel{25} \times \cancel{20}} \\\\

\end{aligned}

Number of boys $= 297$
