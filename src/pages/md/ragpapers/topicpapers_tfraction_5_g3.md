$\dfrac{2}{3} - \dfrac{1}{6}$

<div class='workings'>
<div class='working'>

$
\begin{aligned}
&= \dfrac {2} {3} - \dfrac {1} {6} \\\\
&= \dfrac {2 \times 2} {3 \times 2} - \dfrac {1} {6} \\\\
&= \dfrac {4} {6} - \dfrac {1} {6} \\\\
&= \dfrac {4 - 1} {6} \\\\
&= \dfrac {3} {6} \\\\
&= \dfrac {3} {3 \times 2} \\\\
&= \dfrac {\cancel{3}} {\cancel{3} \times 2} \\\\
&= \dfrac {1} {2}
\end{aligned}
$

</div>

</div>

\begin{aligned}
&= \dfrac{2}{3} \ \text {of} \ 81 \\\\
&= \dfrac {2} {3} \times 81 \\\\
&= \dfrac {2 \times 81} {3} \\\\
&= \dfrac {2 \times 27 \times 3} {3} \\\\
&= \dfrac {2 \times 27 \times \cancel {3}} {\cancel {3}} \\\\
&= 2 \times 27 \\\\
&= 54
\end{aligned}
$
