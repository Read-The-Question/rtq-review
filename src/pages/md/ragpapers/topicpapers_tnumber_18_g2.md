$
\begin{aligned}
4 \times a + 8  &= 44 \\\\
4 a             &= 44 - 8 \\\\
4 a             &= 36 \\\\
a               &= \dfrac {36} {4} \\\\
a               &= \dfrac {9 \times 4} {4} \\\\
a               &= \dfrac {9 \times \cancel {4}} {\cancel {4}} \\\\
a               &= 9
\end{aligned}
