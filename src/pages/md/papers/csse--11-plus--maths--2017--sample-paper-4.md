                        &= \dfrac {96 \times 80} {60} \text{ minutes} \\\\
                        &= \dfrac {12 \times 8 \times 16 \times 5} {12 \times 5}  \\\\
                        &= \dfrac {\cancel{12} \times 8 \times 16 \times \cancel{5}} {\cancel{12} \times \cancel{5}} \\\\
                        &= 128 \text{ minutes} \\\\
                        &= 2 \ \text {hours} \ 8 \ \text {minutes}
\end{aligned}
$
