&= 5 \times 77 \\\\
&= 5 \times 7 \times 11
\end{aligned}
$

\end{aligned}

                                        &= 208 \text {p} \\\\

a                                           &= \dfrac {\cancel {2} \times 14} {\cancel {2}} \\\\

\end{aligned}
